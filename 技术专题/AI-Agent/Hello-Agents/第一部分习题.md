from openai import OpenAI

class OpenAICompatibleClient:
    """
    一个用于调用任何兼容OpenAI接口的LLM服务的客户端
    """

    def __init__(self, model: str, api_key: str, base_url: str):
        self.model = model
        self.client = OpenAI(api_key=api_key, base_url=base_url)

    def generate(self, prompt: str, system_prompt: str) -> str:
        """调用LLM API来生成回应"""
        print("正在调用大语言模型...")
        try:
            messages = [
                {'role': 'system', 'content': system_prompt},
                {'role': 'user', 'content': prompt}
            ]
            # 实际上接口规范是OpenAI定义的，但是LLM模型仍然是各自厂商的
            response = self.client.chat.completions.create(model=self.model, messages=messages, stream=False)
            answer = response.choices[0].message.content
            print('大语言模型响应成功')
            return answer
        except Exception as e:
            print(f'调用LLM API时发生错误：{e}')
            return "错误：调用语言模型服务时出错。"

def get_weather(city: str) -> str:
    """
    通过调用wttr.in API  查询真实的天气信息。
    """
    # API端点，我们请求JSON格式的数据
    url = f'https://wttr.in/{city}?format=j1&lang=zh-cn'

    try:
        # 发起网络请求
        response = requests.get(url)
        # 检查响应状态码是否为200(成功)
        response.raise_for_status()
        # 解析返回的JSON数据
        data = response.json()

        # 提取当前天气状况
        current_condition = data['current_condition'][0]
        weather_desc = current_condition.get('lang_zh-cn', current_condition['weatherDesc'])[0]['value']
        temp_c = current_condition['temp_C']
        return f'{city}当前天气:{weather_desc}，气温{temp_c}摄氏度'
    except requests.exceptions.RequestException as e:
        return f'错误：查询天气时遇到网络问题 - {e}'
    except (KeyError, IndexError) as e:
        return f'错误：解析天气数据失败，可能是城市名称无效 - {e}'

def user_input_pick(user_input: str, llm: Any) -> Dict[str, Any]:
    """
    调用大模型解析用户输入，并返回经过校验的偏好 JSON
    """
    if not isinstance(user_input, str) or not user_input.strip():
        return {}

    # llm 应提供 generate(prompt, system_prompt) 方法，例如 client.py 中的客户端。
    llm_output = llm.generate(
        user_input.strip(),
        system_prompt=USER_INPUT_PICK_PROMPT
    )
    if not isinstance(llm_output, str):
        return {}

    # 兼容模型偶尔返回 ```json ... ``` 的情况，只提取 JSON 对象本身。
    json_match = re.search(r"\{.*\}", llm_output, re.DOTALL)
    if not json_match:
        return {}

    try:
        data = json.loads(json_match.group(0))
    except json.JSONDecodeError as exc:
        return {}

    if not isinstance(data, dict):
        return {}

    like_type = data.get("like_type", [])
    if not isinstance(like_type, list):
        like_type = []
    like_type = list(dict.fromkeys(
        item for item in like_type
        if isinstance(item, str) and item in ALLOWED_LIKE_TYPES
    ))

    budget_range = data.get("budget_range")
    if budget_range not in ALLOWED_BUDGETS:
        budget_range = None

    def clean_list(value):
        if not isinstance(value, list):
            return []
        return list(dict.fromkeys(
            item.strip() for item in value
            if isinstance(item, str) and item.strip()
        ))

    return {
        "like_type": like_type,
        "budget_range": budget_range,
        "hate_spot": clean_list(data.get("hate_spot", [])),
        "dislike_type": clean_list(data.get("dislike_type", [])),
    }

    """

def check_ticket_status(spot_name: str):
    """
    查询指定景点门票是否可购买
    传入多个景点用中文顿号隔开：故宫、天坛、颐和园
    """
    # 模拟票务查询逻辑，实际可对接真实票务接口
    sell_out_list = ["故宫", "八达岭长城"]
    # 分隔多个景点
    spot_list = [s.strip() for s in spot_name.split("、") if s.strip()]
    res_list = []

    for spot_name in spot_list:
        # 1. 记录已推荐、去重
        if spot_name not in spot_ticket_status["already_recommend"]:
            spot_ticket_status["already_recommend"].append(spot_name)

        # 2. 优先判断是否已经标记售罄
        if spot_name in spot_ticket_status["sold_out"]:
            res_list.append(f"查询结果：{spot_name} 已判定门票售罄，无法预约购买")
            continue

        # 3. 优先判断是否再用户讨厌名单
        if spot_name in user_memory["hate_spot"]:
            res_list.append(f"查询结果：{spot_name} 为用户不想去景点，不予推荐")
            continue


        # 4. 模糊匹配是否属于固定售罄景点
        is_sold = any(key in spot_name for key in sell_out_list)
        if is_sold:
            spot_ticket_status["sold_out"].append(spot_name)
            res_list.append(f"查询结果：{spot_name} 当前门票已售罄，无法预约购买")
            # 售罄不加入可推荐列表
            continue

        # 5. 正常可推荐景点
        if spot_name not in spot_ticket_status["can_recommend"]:
            spot_ticket_status["can_recommend"].append(spot_name)

        res_list.append(f"查询结果：{spot_name} 门票充足，正常可购买游玩")

    return "\n".join(res_list)

    # 用程序补充“全部拒绝”范围，避免模型只返回其中一个景点。
    all_rejected = bool(re.search(
        r"都不喜欢|全部(?:不喜欢|不要|不感兴趣)|全都不要|这些都不要|刚刚推荐的(?:内容)?都(?:不喜欢|不感兴趣)",
        feedback_text,
    ))
    if all_rejected:
        attitude = "dislike"
        is_interested = False
        disliked_content = list(dict.fromkeys(disliked_content + mentioned_spots))

    # 补充“只拒绝部分景点”的范围，例如“天坛不想去，颐和园可以”。
    negative_words = r"不喜欢|不想去|不要|没意思|不感兴趣|去过了|太贵"
    feedback_clauses = re.split(r"[，,。；;！？!?\n]", feedback_text)
    for spot in mentioned_spots:
        is_spot_rejected = any(
            spot in clause and re.search(negative_words, clause)
            for clause in feedback_clauses
        )
        if is_spot_rejected and spot not in disliked_content:
            disliked_content.append(spot)

    if disliked_content:
        attitude = "dislike"
        is_interested = False

    # 如果模型识别出了推荐景点，具体拒绝结果只保留其中的景点名称。
    if mentioned_spots and disliked_content:
        disliked_content = [
            spot for spot in disliked_content
            if spot in mentioned_spots
        ]

    # 没有负面语义时，不保留模型误提取的不喜欢内容。
    if attitude != "dislike":
        disliked_content = []
        disliked_type = []

    reason = data.get("reason", "")
    if not isinstance(reason, str) or not reason.strip():
        reason = "用户态度不明确" if attitude == "unknown" else ""

    return {
        "attitude": attitude,
        "is_interested": is_interested,
        "disliked_content": disliked_content,
        "disliked_type": disliked_type,
        "reason": reason.strip(),
    }

def user_think_operate(user_think_input: str, llm: Any) -> bool:
    """
    用LLM判断用户的想法，通过语义决定用户是否不喜欢/拒绝
    1. 提取拒绝景点加入黑名单
    2. 累积拒绝次数
    3. 满3次自动开启策略调整
    """
    global refuse_count, is_adjust_strategy

    data = parse_user_feedback(user_think_input, llm)

    if not data:
        return True

    if data["attitude"] == LIKE or data["attitude"] == UNKNOWN:
        refuse_count = 0
        is_adjust_strategy = False
        return True
    else:
        refuse_count += 1
        print(f'⚠️ 用户已连续拒绝推荐次数：{refuse_count}')

        # 将推荐的景点(can_recommend)添加到hate_spot中
        # 提取用户明确拒绝的景点，添加到hate_spot中，在can_recommend中去除
        for spot in spot_ticket_status["can_recommend"]:
            if data["disliked_content"]:
                if spot in data["disliked_content"]:
                    spot_ticket_status["can_recommend"].remove(spot)
                    print(f"✅ 已将【{spot}】从推荐景点中删除")
                    if spot not in user_memory["hate_spot"]:
                        user_memory["hate_spot"].append(spot)
                        print(f"✅ 已将【{spot}】加入游玩黑名单")

        # 补充不喜欢的风格
        if data["disliked_type"]:
            user_memory["dislike_type"].extend(data["disliked_type"])
            print(f"✅ 已将【{data['disliked_type']}】补充到不喜欢的风格中")

    # 连续拒绝满3次，切换推荐策略
    if refuse_count >= 3 and not is_adjust_strategy:
        is_adjust_strategy = True
        refuse_count = 0
        print("🎉 已连续拒绝3次推荐，智能体开始调整整体推荐策略！")
    else:
        is_adjust_strategy = False

    return False

## 尽管大语言模型驱动的智能体系统展现出了强大的能力，但它们仍然存在诸多局限。请分析以下问题

- 为什么智能体或智能体系统有时会产生"幻觉"（生成看似合理但实际错误的信息）？
- 在 1.3 节的案例中，我们设置了最大循环次数为 5 次。如果没有这个限制，智能体可能会陷入什么问题？
- 如何评估一个智能体的"智能"程度？仅使用准确率指标是否足够？
  
  
一、智能体产生幻觉的原因

1. 训练数据缺陷：训练数据存在错误、过时、矛盾内容，模型被动学习错误知识，输出虚假信息。
2. 知识边界受限：模型存在知识盲区，面对未知内容，会依靠语言逻辑编造通顺内容填补空缺。
3. 逻辑推理不足：仅擅长语言拼接，缺乏事实核验与深度事实推理，只追求语句通顺而非内容真实。
4. 任务指令偏差：用户指令模糊、目标不明确，智能体容易偏离事实，主观臆断作答。
5. 上下文记忆混乱：多轮对话中上下文信息错乱，混淆事实关系，生成错误结论。


二、无最大循环次数限制带来的问题

若无循环次数限制，智能体极易陷入无限循环：

1. 反复重复相同思考、相同操作，无法终止任务。
2. 在错误路径里持续迭代纠错，迟迟无法得出最终结果。
3. 大量消耗算力、网络与运行资源，造成系统卡顿、程序卡死。
4. 任务无法正常收尾，出现死循环，严重影响整体运行稳定性。
   

三、智能程度评估 + 准确率是否足够

1. 评估智能体智能程度维度
   
- 任务完成准确率：结果是否正确达标
- 规划推理能力：能否合理拆分任务、制定流程
- 环境感知与适配能力：能否根据场景变化调整策略
- 工具调用与协作能力：是否会灵活调用工具、配合协作
- 抗干扰与纠错能力：发现错误能否自主修正
- 响应效率与资源消耗：完成任务速度、资源占用情况
- 泛化能力：能否处理从未见过的全新场景

2. 仅用准确率远远不够

准确率只能衡量最终结果对错，无法体现智能体思考逻辑、自主规划、动态适配、自主优化等核心智能特质，也不能判断它是否具备真正自主决策与变通能力，评价维度过于片面单一。
